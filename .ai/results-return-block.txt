  const publishedLabel = (() => {
    if (!selected) return '';
    const title = selected.payload?.title || selected.payload?.body.slice(0, 42) || '已发布内容';
    const at = selected.publication.publishedAt ? new Date(selected.publication.publishedAt) : null;
    const atLabel = at ? `${at.getMonth() + 1}/${at.getDate()} ${String(at.getHours()).padStart(2, '0')}:${String(at.getMinutes()).padStart(2, '0')}` : '时间未知';
    return `《${title}》 · ${platformNames[selected.publication.platform]} · 发布于 ${atLabel}`;
  })();
  return <section className="workflow-page results-page">
    <header className="page-heading today-heading">
      <div><h1>结果</h1>{selected && <p>{publishedLabel}</p>}</div>
      <div className="heading-actions">
        {published.length > 1 && <select className="results-switcher" aria-label="切换已发布内容" value={selected?.publication.id ?? ''} onChange={(event) => setSelectedId(event.target.value)}>{published.map((item) => <option key={item.publication.id} value={item.publication.id}>{item.payload?.title || item.payload?.body.slice(0, 42) || item.publication.id}</option>)}</select>}
        {selected?.publication.platform === 'x' && <button className="secondary-button" disabled={busy} onClick={() => void capturePublication()}>{busy ? '处理中…' : '采集指标'}</button>}
        <button className="primary-button" disabled={busy || !selected || !snapshots.length || locked} onClick={() => void startPiReview()}>{busy ? 'Pi 处理中…' : '写复盘'}</button>
      </div>
    </header>
    {statusText && <p className="task-status" data-running={busy ? 'true' : 'false'}>{statusText}</p>}
    {!selected && <section className="empty-state"><h2>还没有可复盘内容</h2><p>取得真实发布地址后，这里会形成指标与复盘。</p></section>}
    {selected && <>
      {cardKeys.length > 0 && <div className="metric-cards">{cardKeys.map((key) => {
        const latest = snapshots.find((snap) => snap.normalized[key]?.status === 'value')?.normalized[key];
        const delta = deltaOf(key);
        return <div className="metric-card" key={key}>
          <div className="stat-label">{metricLabel(key)}</div>
          <div className="metric-value">{latest?.status === 'value' ? Number(latest.value).toLocaleString('zh-CN') : '—'}</div>
          <div className={`metric-delta${delta === null ? '' : delta >= 0 ? ' up' : ' down'}`}>{delta === null ? '单次采集' : `${delta >= 0 ? '▲ +' : '▼ '}${delta.toLocaleString('zh-CN')}`}</div>
          <div className="spark">{sparkOf(key).map((height, index, bars) => <i key={index} className={index >= bars.length - 2 ? 'hot' : ''} style={{ height: `${height}%` }}/>)}</div>
        </div>;
      })}</div>}
      {snapshots.length > 0 && cardKeys.length > 0 ? <>
        <p className="eyebrow">指标快照 · {chronological.map((snap) => windowLabel(snap.scheduledFor)).join(' / ')}</p>
        <div className="results-panel">
          <table className="snap-table"><thead><tr><th>计划时间</th><th>实际采集</th>{cardKeys.map((key) => <th key={key}>{metricLabel(key)}</th>)}<th>字段状态</th></tr></thead><tbody>
            {chronological.map((snap) => {
              const missing = cardKeys.filter((key) => snap.normalized[key]?.status !== 'value').length;
              return <tr key={snap.id}>
                <td className="num faint">{windowLabel(snap.scheduledFor)}</td>
                <td className="num">{new Date(snap.capturedAt).toLocaleString('zh-CN')}</td>
                {cardKeys.map((key) => <td className="num" key={key}>{snap.normalized[key]?.status === 'value' ? Number(snap.normalized[key]?.value).toLocaleString('zh-CN') : '—'}</td>)}
                <td>{missing === 0 ? <span className="pill-status green"><span className="dot"/>全部有值</span> : <span className="pill-status gray">{missing} 项不可见</span>}</td>
              </tr>;
            })}
          </tbody></table>
        </div>
      </> : <div className="results-panel results-empty-hint">还没有该内容的指标快照。采集后形成数据驱动复盘；没有快照时复盘只能保持草稿。</div>}
      <p className="eyebrow">复盘 · {activeReview ? activeReview.status === 'final' ? `已定稿${activeReview.finalizedAt ? `(${new Date(activeReview.finalizedAt).toLocaleDateString('zh-CN')})` : ''}` : '草稿' : '未创建'}</p>
      {locked && activeReview ? <>
        {activeReview.summary && <p className="review-summary">{activeReview.summary}</p>}
        <div className="review-cols">
          <div className="review-col keep"><h4>✓ 保留 Keep</h4><ul>{activeReview.keep.map((item, index) => <li key={index}>{item}</li>)}</ul></div>
          <div className="review-col stop"><h4>✕ 停止 Stop</h4><ul>{activeReview.stop.map((item, index) => <li key={index}>{item}</li>)}</ul></div>
          <div className="review-col change"><h4>↻ 改变 Change</h4><ul>{activeReview.change.map((item, index) => <li key={index}>{item}</li>)}</ul></div>
        </div>
      </> : <div className="results-panel review-edit-panel">
        <label className="review-field"><span>摘要</span><textarea disabled={busy} value={summary} onChange={(event) => setSummary(event.target.value)} placeholder="一句话说明这次内容的结果判断" /></label>
        <div className="review-grid editable">
          <label><span>✓ 保留 Keep</span><textarea disabled={busy} value={keepText} onChange={(event) => setKeepText(event.target.value)} placeholder="每行一条" /></label>
          <label><span>✕ 停止 Stop</span><textarea disabled={busy} value={stopText} onChange={(event) => setStopText(event.target.value)} placeholder="每行一条" /></label>
          <label><span>↻ 改变 Change</span><textarea disabled={busy} value={changeText} onChange={(event) => setChangeText(event.target.value)} placeholder="每行一条" /></label>
        </div>
        <div className="review-finding-row">
          <label className="review-field"><span>方法结论标题</span><input disabled={busy} value={findingTitle} onChange={(event) => setFindingTitle(event.target.value)} placeholder="例如：视频封面要先给结论" /></label>
          <label className="review-field"><span>内容</span><textarea disabled={busy} value={findingBody} onChange={(event) => setFindingBody(event.target.value)} placeholder="后续方案可引用这条结论" /></label>
        </div>
        <div className="heading-actions">
          <button className="secondary-button" disabled={busy || !latestSnapshot} onClick={() => void saveCurrentReview('draft')}>保存草稿</button>
          <button className="primary-button" disabled={busy || !latestSnapshot} onClick={() => void saveCurrentReview('final')}>定稿复盘</button>
        </div>
      </div>}
      {(activeReview?.findings?.length || backlinks.length > 0) && <p className="results-footnote">{activeReview?.findings?.length ? `${activeReview.findings.length} 条方法结论已保存` : ''}{backlinks.length ? ` · 引用链：复盘 → 方法结论 → ${backlinks.length} 个后续方案` : activeReview?.findings?.length ?
